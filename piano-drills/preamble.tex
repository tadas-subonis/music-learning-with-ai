% ---------- piano keyboard ----------
% \kbd[scale]{octaves}{marks: semitone/label,...}{roots: semitone,...}
% semitone 0 = the C at the left edge
\newcommand{\kbd}[4][0.3]{%
\begin{tikzpicture}[x=#1cm,y=#1cm]
  \pgfmathtruncatemacro\nw{7*#2-1}
  \foreach \i in {0,...,\nw} {\draw[line width=0.45pt] (\i,0) rectangle ++(1,4.6);}
  \draw[line width=0.45pt] (7*#2,0) rectangle ++(1,4.6);
  \foreach \o in {1,...,#2} {\foreach \b in {1,2,4,5,6} {
     \fill ({7*(\o-1)+\b-0.32},1.75) rectangle ++(0.64,2.85);}}
  \foreach \s/\l in {#3} {
    \pgfmathtruncatemacro\r{mod(\s,12)}
    \pgfmathtruncatemacro\o{int(\s/12)}
    \pgfmathtruncatemacro\w{{0,0,1,1,2,3,3,4,4,5,5,6}[\r]}
    \pgfmathtruncatemacro\bk{{0,1,0,1,0,0,1,0,1,0,1,0}[\r]}
    \ifnum\bk=0
      \fill (7*\o+\w+0.5,0.85) circle (0.43);
      \node[text=white,font=\bfseries\fontsize{7}{7}\selectfont] at (7*\o+\w+0.5,0.85) {\l};
    \else
      \fill[white] (7*\o+\w+1,2.35) circle (0.3);
      \node[text=black,font=\bfseries\fontsize{6}{6}\selectfont] at (7*\o+\w+1,2.35) {\l};
    \fi
  }
  \foreach \s in {#4} {
    \pgfmathtruncatemacro\r{mod(\s,12)}
    \pgfmathtruncatemacro\o{int(\s/12)}
    \pgfmathtruncatemacro\w{{0,0,1,1,2,3,3,4,4,5,5,6}[\r]}
    \pgfmathtruncatemacro\bk{{0,1,0,1,0,0,1,0,1,0,1,0}[\r]}
    \pgfmathsetmacro\xx{\bk==0 ? 7*\o+\w+0.5 : 7*\o+\w+1}
    \fill (\xx-0.4,-0.95)--(\xx+0.4,-0.95)--(\xx,-0.25)--cycle;
  }
\end{tikzpicture}}
